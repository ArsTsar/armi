import { NextRequest, NextResponse } from "next/server";
import ZAI from "z-ai-web-dev-sdk";

export const runtime = "nodejs";
export const maxDuration = 300;

// Расшифровка аудио (речь → текст). Принимает multipart-форму с полем audio.
export async function POST(req: NextRequest) {
  try {
    const form = await req.formData();
    const file = form.get("audio");
    if (!(file instanceof Blob)) {
      return NextResponse.json({ error: "Нет аудиофайла" }, { status: 400 });
    }
    if (file.size > 25 * 1024 * 1024) {
      return NextResponse.json({ error: "Файл больше 25 МБ — обрежь запись или загрузи по частям" }, { status: 413 });
    }

    const buf = Buffer.from(await file.arrayBuffer());
    const base64 = buf.toString("base64");

    const zai = await ZAI.create();
    const res = await zai.audio.asr.create({ file_base64: base64 });
    const text = (res.text ?? "").trim();
    if (!text) {
      return NextResponse.json({ error: "Речь не распознана" }, { status: 422 });
    }
    return NextResponse.json({ text });
  } catch (e) {
    const msg = e instanceof Error ? e.message : "Ошибка расшифровки";
    return NextResponse.json({ error: msg }, { status: 500 });
  }
}
